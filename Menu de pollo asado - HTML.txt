  <script>
    // Config and order state
    const defaultConfig = {
      restaurant_name: 'THE CHICKEN HOUSE',
      tagline: 'Pollo asado a la perfección',
      background_color: '#FFED00',
      surface_color: '#ffffff',
      text_color: '#1a1a1a',
      accent_color: '#EF2121',
      secondary_text_color: '#666'
    };
    
    let config = { ...defaultConfig };
    let orderItems = [];
    let orderIdCounter = 0;

    // Order functions
    function addToOrder(name, priceMXN, priceUSD, category) {
      const categoryColors = {
        'pollo-clasico': '#EF2121',
        'pollo-sabor': '#FF6B00',
        'extra': '#8B4513'
      };
      
      // Verificar si el item ya existe
      const existingItem = orderItems.find(item => item.name === name);
      
      if (existingItem) {
        existingItem.quantity = (existingItem.quantity || 1) + 1;
      } else {
        orderItems.push({
          id: ++orderIdCounter,
          name: name,
          priceMXN: priceMXN,
          priceUSD: priceUSD,
          category: category,
          color: categoryColors[category],
          quantity: 1
        });
      }
      
      renderOrder();
    }

    function removeFromOrder(id) {
      const item = orderItems.find(item => item.id === id);
      if (item && item.quantity > 1) {
        item.quantity--;
      } else {
        orderItems = orderItems.filter(item => item.id !== id);
      }
      renderOrder();
    }

    function clearOrder() {
      orderItems = [];
      renderOrder();
    }

    function sendToWhatsApp() {
      if (orderItems.length === 0) {
        alert('Por favor agrega items a tu pedido');
        return;
      }

      let mensaje = 'Buenas tardes,%0AQuiero ordenar:%0A';
      
      orderItems.forEach(item => {
        const qty = item.quantity;
        const name = item.name;
        
        // Singulares y plurales para extras
        if (name === 'Salchicha Asada') {
          mensaje += qty === 1 ? '- una salchicha%0A' : `- ${qty} salchichas%0A`;
        } else if (name === 'Papa Asada') {
          mensaje += qty === 1 ? '- una papa%0A' : `- ${qty} papas%0A`;
        } else if (name === 'Arroz') {
          mensaje += qty === 1 ? '- un arroz%0A' : `- ${qty} arroces%0A`;
        } else {
          // Para pollos
          mensaje += qty === 1 ? `- Un paquete de ${name}%0A` : `- ${qty} paquetes de ${name}%0A`;
        }
      });

      mensaje += 'Gracias';

      const whatsappUrl = `https://wa.me/526566423548?text=${mensaje}`;
      window.open(whatsappUrl, '_blank');
    }

    function renderOrder() {
      const container = document.getElementById('order-items');
      const emptyMessage = document.getElementById('empty-order');
      const summary = document.getElementById('order-summary');
      
      if (orderItems.length === 0) {
        container.innerHTML = `<p id="empty-order" class="text-sm text-center py-8 font-medium" style="color: #999;">Toca un plato para agregarlo</p>`;
        summary.classList.add('hidden');
        return;
      }
      
      summary.classList.remove('hidden');
      
      container.innerHTML = orderItems.map(item => `
        <div class="order-item flex items-center justify-between p-3 rounded-lg shadow" style="background-color: #f9f9f9; border: 2px solid ${item.color};">
          <div class="flex items-center gap-3 flex-1">
            <div class="w-2 h-10 rounded-full" style="background-color: ${item.color};"></div>
            <div class="flex-1">
              <p class="text-sm font-bold" style="color: ${config.text_color};">${item.name}</p>
              <p class="text-xs font-medium" style="color: ${config.secondary_text_color};">$${item.priceMXN} MXN / $${item.priceUSD.toFixed(2)} USD</p>
            </div>
          </div>
          <div class="flex items-center gap-2">
            <button onclick="removeFromOrder(${item.id})" class="w-6 h-6 rounded-full flex items-center justify-center transition-colors font-bold text-sm" style="background-color: #f0f0f0; color: #999;" onmouseover="this.style.backgroundColor='#EF2121'; this.style.color='#fff'" onmouseout="this.style.backgroundColor='#f0f0f0'; this.style.color='#999'">−</button>
            <span class="w-6 text-center font-bold text-sm" style="color: ${config.text_color};">${item.quantity}</span>
            <button onclick="addToOrder('${item.name.replace(/'/g, "\\'")}', ${item.priceMXN}, ${item.priceUSD}, '${item.category}')" class="w-6 h-6 rounded-full flex items-center justify-center transition-colors font-bold text-sm" style="background-color: #f0f0f0; color: #999;" onmouseover="this.style.backgroundColor='#EF2121'; this.style.color='#fff'" onmouseout="this.style.backgroundColor='#f0f0f0'; this.style.color='#999'">+</button>
          </div>
        </div>
      `).join('');
      
      const totalMXN = orderItems.reduce((sum, item) => sum + (item.priceMXN * item.quantity), 0);
      const totalUSD = orderItems.reduce((sum, item) => sum + (item.priceUSD * item.quantity), 0);
      
      document.getElementById('total').textContent = `$${totalMXN.toFixed(2)}`;
      document.getElementById('total-usd').textContent = `$${totalUSD.toFixed(2)}`;
    }

    // Element SDK implementation
    async function onConfigChange(newConfig) {
      config = { ...config, ...newConfig };
      
      document.getElementById('restaurant-name').textContent = config.restaurant_name || defaultConfig.restaurant_name;
      document.getElementById('tagline').textContent = config.tagline || defaultConfig.tagline;
    }

    function mapToCapabilities(config) {
      return {
        recolorables: [
          {
            get: () => config.background_color || defaultConfig.background_color,
            set: (value) => { config.background_color = value; window.elementSdk.setConfig({ background_color: value }); }
          },
          {
            get: () => config.surface_color || defaultConfig.surface_color,
            set: (value) => { config.surface_color = value; window.elementSdk.setConfig({ surface_color: value }); }
          },
          {
            get: () => config.text_color || defaultConfig.text_color,
            set: (value) => { config.text_color = value; window.elementSdk.setConfig({ text_color: value }); }
          },
          {
            get: () => config.accent_color || defaultConfig.accent_color,
            set: (value) => { config.accent_color = value; window.elementSdk.setConfig({ accent_color: value }); }
          },
          {
            get: () => config.secondary_text_color || defaultConfig.secondary_text_color,
            set: (value) => { config.secondary_text_color = value; window.elementSdk.setConfig({ secondary_text_color: value }); }
          }
        ],
        borderables: [],
        fontEditable: undefined,
        fontSizeable: undefined
      };
    }

    function mapToEditPanelValues(config) {
      return new Map([
        ['restaurant_name', config.restaurant_name || defaultConfig.restaurant_name],
        ['tagline', config.tagline || defaultConfig.tagline]
      ]);
    }

    // Initialize
    if (window.elementSdk) {
      window.elementSdk.init({
        defaultConfig,
        onConfigChange,
        mapToCapabilities,
        mapToEditPanelValues
      });
